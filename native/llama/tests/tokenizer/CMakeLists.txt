cmake_minimum_required(VERSION 3.22)
project(skein_tokenizer_parity CXX C)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(UPSTREAM "${REPO_ROOT}/third_party/llama.cpp")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
add_subdirectory("${UPSTREAM}" "${CMAKE_CURRENT_BINARY_DIR}/llama.cpp" EXCLUDE_FROM_ALL)
include("${REPO_ROOT}/native/llama/tokenizer-overlay.cmake")
skein_tokenizer_overlay("${UPSTREAM}" "${CMAKE_CURRENT_BINARY_DIR}/tokenizer-src")
add_executable(tokenizer-parity tokenizer-parity.cpp)
target_include_directories(tokenizer-parity PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tokenizer-src")
target_link_libraries(tokenizer-parity PRIVATE llama)
target_compile_features(tokenizer-parity PRIVATE cxx_std_17)
