# -----------------------------------------------------------------------------
# Skein native SQLite: SQLCipher (OpenSSL-backed) + sqlite-vec + FTS5, statically
# linked into a single Android shared object `libskein_sqlite.so`.
#
# The vendored androidx.sqlite JNI driver (skein-e2ki) will link this .so and
# expose it to Kotlin as `SkeinSQLiteDriver` (E2.I1).
#
# Recipe reference: `native/sqlite/README.md` and `docs/SPIKE_E0_I7_RESULTS.md`.
# Reproducibility flags mirror `E1.I4`:
#   * `-ffile-prefix-map` strips absolute build paths from __FILE__
#   * `-Wl,--build-id=none` drops the ELF build-id note
#   * `-fno-ident` removes `.comment` toolchain strings
#   * `-Wdate-time -Werror=date-time` fails the build on any __DATE__/__TIME__
#   * `SOURCE_DATE_EPOCH` honoured by every subordinate build (OpenSSL, ar, ld)
#   * OpenSSL is configured with a fixed synthetic `--prefix`/`--openssldir`
#     and installed via `DESTDIR`, so no checkout path is compiled into
#     libcrypto.a (skein-ej1b) — see section 3.
# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# 0. Amalgamation provenance pins + integrity check
# -----------------------------------------------------------------------------
# The SQLCipher amalgamation (amalgamation/sqlite3.c, sqlite3.h, sqlite3ext.h)
# is checked into git rather than fetched/regenerated by this CMake build --
# see docs/design/AMALGAMATION_POLICY.md for the full rationale (decided in
# skein-2lq9). These pins are the single source of truth for "what upstream
# commit does the checked-in amalgamation correspond to"; they must match
# native/sqlite/sqlcipher.sha256 and native/sqlite/verify_amalgamation.sh.
# Bumping SQLCipher means updating all four together (this block,
# sqlcipher.sha256, the three amalgamation files, and
# amalgamation/SHA256SUMS.txt) in one reviewed commit -- see README.md §3
# ("Generate the SQLCipher amalgamation") for the regeneration recipe.
set(SQLCIPHER_PINNED_VERSION    "4.17.0")
set(SQLCIPHER_PINNED_GIT_TAG    "v4.17.0")
set(SQLCIPHER_PINNED_GIT_COMMIT "f9788efa8ac4dfed75c03e4756b1666a1d0845da")
# Date this amalgamation was generated and confirmed byte-identical to the
# pinned source via `native/sqlite/verify_amalgamation.sh --regenerate`.
set(AMALGAMATION_GENERATED_DATE "2026-09-20")

set(_AMALGAMATION_DIR "${SKEIN_SQLITE_ROOT}/amalgamation")
set(_AMALGAMATION_SHA256SUMS "${_AMALGAMATION_DIR}/SHA256SUMS.txt")

# Fast (<1s) tamper/drift check: hash the three checked-in files and compare
# against SHA256SUMS.txt on every configure (every build, not just release
# tags). The slow check -- actually re-running SQLCipher's configure/make
# against the pinned tag and diffing the result -- lives in the release-tag
# CI job (.github/workflows/reproducible-build.yml, job
# "amalgamation-integrity") via this same script's `--regenerate` mode,
# since it needs tclsh/make/network and isn't cheap enough for every build.
if(NOT EXISTS "${_AMALGAMATION_SHA256SUMS}")
  message(FATAL_ERROR
    "Missing ${_AMALGAMATION_SHA256SUMS}. See docs/design/AMALGAMATION_POLICY.md.")
endif()

file(STRINGS "${_AMALGAMATION_SHA256SUMS}" _amalgamation_sha_lines)
foreach(_amalgamation_sha_line IN LISTS _amalgamation_sha_lines)
  if(_amalgamation_sha_line MATCHES "^([0-9a-fA-F]+)[ \t]+(.+)$")
    set(_expected_hash "${CMAKE_MATCH_1}")
    set(_expected_file "${CMAKE_MATCH_2}")
    set(_actual_path "${_AMALGAMATION_DIR}/${_expected_file}")
    if(NOT EXISTS "${_actual_path}")
      message(FATAL_ERROR
        "Amalgamation integrity check: ${_expected_file} is listed in "
        "SHA256SUMS.txt but is missing from ${_AMALGAMATION_DIR}.")
    endif()
    file(SHA256 "${_actual_path}" _actual_hash)
    string(TOLOWER "${_expected_hash}" _expected_hash_lc)
    if(NOT _actual_hash STREQUAL _expected_hash_lc)
      message(FATAL_ERROR
        "Amalgamation integrity check FAILED for ${_expected_file}:\n"
        "  expected sha256 ${_expected_hash_lc}\n"
        "  actual   sha256 ${_actual_hash}\n"
        "amalgamation/${_expected_file} does not match "
        "amalgamation/SHA256SUMS.txt -- this means it was modified without "
        "updating the hash file (tampering, a bad merge, or a partial "
        "SQLCipher bump). Regenerate deliberately per "
        "native/sqlite/README.md and docs/design/AMALGAMATION_POLICY.md, or "
        "restore the file from git if this wasn't intentional.")
    endif()
  endif()
endforeach()
message(STATUS "Amalgamation integrity check passed (SQLCipher ${SQLCIPHER_PINNED_VERSION} / ${SQLCIPHER_PINNED_GIT_COMMIT})")

# -----------------------------------------------------------------------------
# 3. Fetch pinned OpenSSL tarball (verified sha256)
# -----------------------------------------------------------------------------
set(OPENSSL_VERSION "3.5.4")
set(OPENSSL_SHA256  "967311f84955316969bdb1d8d4b983718ef42338639c621ec4c34fddef355e99")
set(OPENSSL_URL     "https://github.com/openssl/openssl/releases/download/openssl-${OPENSSL_VERSION}/openssl-${OPENSSL_VERSION}.tar.gz")

set(OPENSSL_SRC_DIR      "${SKEIN_SQLITE_ROOT}/third_party/openssl-${OPENSSL_VERSION}")
set(OPENSSL_TARBALL      "${SKEIN_SQLITE_ROOT}/third_party/openssl-${OPENSSL_VERSION}.tar.gz")

# Per-ABI build tree, kept under the CMake build dir so `./gradlew clean` wipes it.
set(OPENSSL_BUILD_DIR    "${CMAKE_CURRENT_BINARY_DIR}/openssl-${OPENSSL_VERSION}-${ANDROID_ABI}")

# Reproducibility (skein-ej1b): OpenSSL compiles its `--prefix`-derived
# directories into crypto/info.c as the string constants OPENSSLDIR,
# ENGINESDIR and MODULESDIR. Those objects end up in libcrypto.a and get
# linked into libskein_sqlite.so, so a build-tree --prefix leaks the absolute
# checkout path (…/core/vault/.cxx/Debug/<agp-hash>/…) into the shipped .so
# and makes its sha256 depend on where the repo happens to live. They are
# dead strings for us — `no-dso no-engine no-legacy` means nothing is ever
# loaded from them and SQLCipher never reads a config file — so we bake a
# fixed, synthetic prefix instead and stage the real install tree underneath
# it with DESTDIR (see INSTALL_COMMAND below). The *baked* strings are then
# constant on every machine, while OPENSSL_PREFIX stays a build-tree path for
# linking and header lookup.
set(OPENSSL_BAKED_PREFIX "/skein-openssl")
set(OPENSSL_DESTDIR      "${CMAKE_CURRENT_BINARY_DIR}/openssl-${OPENSSL_VERSION}-${ANDROID_ABI}-install")
set(OPENSSL_PREFIX       "${OPENSSL_DESTDIR}${OPENSSL_BAKED_PREFIX}")
set(OPENSSL_LIBCRYPTO    "${OPENSSL_PREFIX}/lib/libcrypto.a")
set(OPENSSL_INCLUDE_DIR  "${OPENSSL_PREFIX}/include")

# Fetch once (per repo). If the tarball is missing or its sha256 doesn't match,
# re-download. This runs at CMake configure time so subsequent builds are fast.
if(NOT EXISTS "${OPENSSL_TARBALL}")
  message(STATUS "Fetching OpenSSL ${OPENSSL_VERSION} tarball (${OPENSSL_URL})…")
  file(DOWNLOAD "${OPENSSL_URL}" "${OPENSSL_TARBALL}"
       EXPECTED_HASH SHA256=${OPENSSL_SHA256}
       SHOW_PROGRESS
       TLS_VERIFY ON
       STATUS _dl_status)
  list(GET _dl_status 0 _dl_code)
  if(NOT _dl_code EQUAL 0)
    file(REMOVE "${OPENSSL_TARBALL}")
    message(FATAL_ERROR "Failed to download OpenSSL: ${_dl_status}")
  endif()
else()
  file(SHA256 "${OPENSSL_TARBALL}" _existing_sha)
  if(NOT _existing_sha STREQUAL "${OPENSSL_SHA256}")
    message(STATUS "OpenSSL tarball sha mismatch — refetching")
    file(DOWNLOAD "${OPENSSL_URL}" "${OPENSSL_TARBALL}"
         EXPECTED_HASH SHA256=${OPENSSL_SHA256}
         SHOW_PROGRESS
         TLS_VERIFY ON)
  endif()
endif()

# Reference extraction under third_party/. It is the tree README.md §4's manual
# recipe and the host harness work in, and it is NOT the input to the per-ABI
# Android builds: since skein-doq0 those extract the pinned tarball themselves
# (see section 4), so anything you leave lying around here cannot reach a .so.
if(NOT EXISTS "${OPENSSL_SRC_DIR}/Configure")
  message(STATUS "Extracting OpenSSL ${OPENSSL_VERSION}…")
  file(ARCHIVE_EXTRACT INPUT "${OPENSSL_TARBALL}"
       DESTINATION "${SKEIN_SQLITE_ROOT}/third_party")
endif()

include(ExternalProject)

# Find the NDK's host prebuilt bin dir so OpenSSL's Configure can locate
# clang/ar/ranlib. r27+ ships only darwin-x86_64/linux-x86_64/windows-x86_64.
# NB: `APPLE`/`WIN32` refer to the *target* platform when cross-compiling;
# `CMAKE_HOST_*` is what we need for host-side toolchain path selection.
if(CMAKE_HOST_APPLE)
  set(_NDK_HOST_TAG "darwin-x86_64")
elseif(CMAKE_HOST_WIN32)
  set(_NDK_HOST_TAG "windows-x86_64")
else()
  set(_NDK_HOST_TAG "linux-x86_64")
endif()
set(_NDK_TOOLCHAIN_BIN "${ANDROID_NDK}/toolchains/llvm/prebuilt/${_NDK_HOST_TAG}/bin")
if(NOT EXISTS "${_NDK_TOOLCHAIN_BIN}")
  message(FATAL_ERROR "NDK toolchain bin dir not found: ${_NDK_TOOLCHAIN_BIN}")
endif()

# SOURCE_DATE_EPOCH: honour an ambient value from the Gradle build (E1.I8), and
# otherwise fall back to a deterministic constant so two clean builds on the
# same machine still produce identical bytes.
if(DEFINED ENV{SOURCE_DATE_EPOCH})
  set(_SOURCE_DATE_EPOCH "$ENV{SOURCE_DATE_EPOCH}")
else()
  # 2026-01-01T00:00:00Z — arbitrary, but stable.
  set(_SOURCE_DATE_EPOCH "1767225600")
endif()

# OpenSSL Configure disables shared, DSO, engine and tests: shaves ~1 MB from
# libcrypto.a and removes the runtime dynamic-provider loader (spec §5 hardening).
#
# `--prefix` / `--openssldir` are deliberately *not* the real install location
# (skein-ej1b): every value derived from them is compiled into libcrypto.a as
# a string literal, so they must be identical on every machine. The real
# staging root is supplied at install time via DESTDIR. `--openssldir` is
# passed explicitly rather than left to default to `<prefix>/ssl` so the
# baked value stays pinned even if a future OpenSSL changes that default.
set(_OPENSSL_CONFIGURE_ARGS
  "${OPENSSL_TARGET}"
  "-D__ANDROID_API__=${_ANDROID_API_LEVEL}"
  no-shared
  no-tests
  no-dso
  no-engine
  no-legacy
  "--prefix=${OPENSSL_BAKED_PREFIX}"
  "--openssldir=${OPENSSL_BAKED_PREFIX}/ssl"
  "--libdir=lib"
)

# `-ffile-prefix-map` for the libcrypto translation units, so `__FILE__` (and
# any debug-info path) cannot carry the per-ABI build tree's absolute path.
#
# This is passed as a *make command-line* `CPPFLAGS` override rather than as a
# Configure argument, and the difference matters (skein-ej1b). Flags handed to
# Configure land in the generated Makefile's `CFLAGS`, and `crypto/buildinf.h`
# is generated from `$(CC) $(LIB_CFLAGS) $(CPPFLAGS_Q)` — so the flag *text*,
# absolute paths and all, gets compiled into crypto/info.c's "compiler: …"
# string and lands right back in the .so, trading one path leak for another.
# `CPPFLAGS` reaches every compile through `LIB_CPPFLAGS`/`DSO_CPPFLAGS`/
# `BIN_CPPFLAGS` but is absent from the buildinf.h recipe, which records the
# frozen `CPPFLAGS_Q` literal instead. Overriding it on the command line
# drops the Makefile's own `CPPFLAGS=-D__ANDROID_API__=…`, so that is restated
# here; `CPPFLAGS_Q` (and therefore the recorded string) is unaffected.
set(_OPENSSL_MAKE_CPPFLAGS
  "CPPFLAGS=-D__ANDROID_API__=${_ANDROID_API_LEVEL} -ffile-prefix-map=${OPENSSL_BUILD_DIR}=/skein/openssl -ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=/skein/build")

# OpenSSL only supports in-tree builds, so each ABI gets its own source tree
# in BINARY_DIR (materialised by the CONFIGURE_COMMAND below) and runs
# ./Configure + make there.
set(_OPENSSL_ENV
  ${CMAKE_COMMAND} -E env
    "ANDROID_NDK_ROOT=${ANDROID_NDK}"
    "PATH=${_NDK_TOOLCHAIN_BIN}:$ENV{PATH}"
    "SOURCE_DATE_EPOCH=${_SOURCE_DATE_EPOCH}"
)

# Helper scripts driving the two steps that must not depend on ambient state:
# a pristine per-ABI source tree, and an ABI sanity check on the result. Both
# are plain CMake script-mode files (no shell), so they behave identically on
# a developer's zsh and on CI's runner.
set(_OPENSSL_EXTRACT_SCRIPT "${SKEIN_SQLITE_ROOT}/cmake/extract_openssl.cmake")
set(_OPENSSL_ABI_GUARD_SCRIPT "${SKEIN_SQLITE_ROOT}/cmake/verify_openssl_abi.cmake")
foreach(_openssl_script "${_OPENSSL_EXTRACT_SCRIPT}" "${_OPENSSL_ABI_GUARD_SCRIPT}")
  if(NOT EXISTS "${_openssl_script}")
    message(FATAL_ERROR "Missing OpenSSL build helper script: ${_openssl_script}")
  endif()
endforeach()

ExternalProject_Add(openssl_ep
  SOURCE_DIR        "${OPENSSL_SRC_DIR}"
  BINARY_DIR        "${OPENSSL_BUILD_DIR}"
  INSTALL_DIR       "${OPENSSL_PREFIX}"
  DOWNLOAD_COMMAND  ""
  UPDATE_COMMAND    ""
  # skein-doq0: BINARY_DIR is (re)created by extracting the sha256-pinned
  # tarball, NOT by copying ${OPENSSL_SRC_DIR}. The shared extraction under
  # third_party/ is a working directory README.md §4 tells you to run
  # ./Configure + make in, and copying it dragged that ABI's .o files,
  # Makefile and configdata.pm into the next ABI's build — `make` then saw
  # them as up to date and installed a libcrypto.a full of foreign-arch
  # objects. Extracting per ABI makes that structurally impossible and pins
  # every ABI's inputs to the exact bytes openssl.sha256 names. Each step is
  # its own CMake `COMMAND` block so ExternalProject can log/rerun them
  # independently. `cmake -E chdir` handles the WORKING_DIRECTORY hop,
  # because ExternalProject_Add's COMMAND blocks don't take a
  # WORKING_DIRECTORY keyword — the extract step chdirs to a directory it is
  # not about to delete.
  CONFIGURE_COMMAND
    ${CMAKE_COMMAND} -E chdir "${SKEIN_SQLITE_ROOT}"
      ${CMAKE_COMMAND}
        "-DOPENSSL_TARBALL=${OPENSSL_TARBALL}"
        "-DOPENSSL_SHA256=${OPENSSL_SHA256}"
        "-DOPENSSL_ARCHIVE_ROOT=openssl-${OPENSSL_VERSION}"
        "-DOPENSSL_BUILD_DIR=${OPENSSL_BUILD_DIR}"
        -P "${_OPENSSL_EXTRACT_SCRIPT}"
  COMMAND
    ${CMAKE_COMMAND} -E chdir "${OPENSSL_BUILD_DIR}"
      ${_OPENSSL_ENV} perl ./Configure ${_OPENSSL_CONFIGURE_ARGS}
  BUILD_COMMAND
    ${CMAKE_COMMAND} -E chdir "${OPENSSL_BUILD_DIR}"
      ${_OPENSSL_ENV} make -j "${_OPENSSL_MAKE_CPPFLAGS}" build_libs
  # Install, then immediately prove the installed archive is for this ABI.
  # Without the guard a wrong-arch libcrypto.a only surfaces minutes later as
  # an opaque `ld.lld: error: ... is incompatible with aarch64linux`.
  # DESTDIR relocates the install under the per-ABI build tree without touching
  # the prefix baked into libcrypto.a (skein-ej1b). It has to be a *make
  # command-line* variable: OpenSSL's generated Makefile contains its own
  # `DESTDIR=` assignment, which would otherwise win over the environment and
  # send `make install_dev` at the literal /skein-openssl.
  INSTALL_COMMAND
    ${CMAKE_COMMAND} -E chdir "${OPENSSL_BUILD_DIR}"
      ${_OPENSSL_ENV} make "DESTDIR=${OPENSSL_DESTDIR}" "${_OPENSSL_MAKE_CPPFLAGS}" install_dev
  COMMAND
    ${CMAKE_COMMAND}
      "-DLIBCRYPTO=${OPENSSL_LIBCRYPTO}"
      "-DLLVM_AR=${_NDK_TOOLCHAIN_BIN}/llvm-ar"
      "-DLLVM_READELF=${_NDK_TOOLCHAIN_BIN}/llvm-readelf"
      "-DTARGET_ABI=${ANDROID_ABI}"
      "-DEXPECTED_MACHINE=${OPENSSL_EXPECTED_ELF_MACHINE}"
      "-DSCRATCH_DIR=${CMAKE_CURRENT_BINARY_DIR}"
      -P "${_OPENSSL_ABI_GUARD_SCRIPT}"
  BUILD_BYPRODUCTS "${OPENSSL_LIBCRYPTO}"
  USES_TERMINAL_BUILD TRUE
  LOG_CONFIGURE ON
  LOG_BUILD ON
  LOG_INSTALL ON
)

# -----------------------------------------------------------------------------
# 5a. Linker version script (skein-6rwv)
#
# `-fvisibility=hidden` above hides every symbol in a compiled object with
# STV_HIDDEN, which a linker cannot re-expose via a version script. To let a
# JNI shim (skein-e2ki) link/dlsym the SQLite / SQLCipher / vec / skein init
# entry points, we override `SQLITE_API` / `SQLITE_VEC_API` so the specific
# public declarations we care about are emitted with default visibility, and
# then use a generated version script's `local: *;` clause to trim every
# other default-visible symbol back down to local. Result: only the whitelist
# below is exported; every internal helper (including anything a future
# contributor accidentally marks default-visible) still stays private.
#
# The whitelist is generated at CMake configure time by grepping
# amalgamation/sqlite3.h for `^SQLITE_API` and appending SQLCipher / vec /
# our own init entry points. Names are sorted so the script bytes (and the
# .so bytes) reproduce identically across clean builds.
# -----------------------------------------------------------------------------
# The vendored translation units (SQLCipher amalgamation, sqlite-vec) must
# emit their symbols with default visibility so the version script's whitelist
# can name them. A version script cannot re-expose an STV_HIDDEN symbol, and
# `-fvisibility=hidden` above would otherwise hide every SQLITE_API /
# SQLITE_VEC_API declaration (both `SQLITE_API` in sqlite3.h and
# `SQLITE_VEC_API` in sqlite-vec.h expand to empty on Android). Override
# `-fvisibility=hidden` for these two TUs specifically — every other TU we
# compile (skein_extra_init.c and anything future) keeps hidden as its
# default, and the version script's `local: *;` clause is what actually
# trims the exported set down to the whitelist. A last-wins `-fvisibility`
# on the command line is how clang picks between the two flags.
set_source_files_properties(
  "${SKEIN_SQLITE_ROOT}/amalgamation/sqlite3.c"
  "${SQLITE_VEC_DIR}/sqlite-vec.c"
  PROPERTIES COMPILE_OPTIONS "-fvisibility=default"
)

set(_SKEIN_EXPORTS_LD "${CMAKE_CURRENT_BINARY_DIR}/skein_sqlite.exports.ld")

# Collect the SQLite public API from the amalgamation header. Each `^SQLITE_API`
# line is one of three shapes:
#   1. `SQLITE_API <type> IDENT(<args>);`     — function
#   2. `SQLITE_API <type> IDENT(`             — function whose args wrap
#   3. `SQLITE_API SQLITE_EXTERN <type> IDENT[];` or `... IDENT;` — variable
# In every case the identifier is the last whitespace/`*`-separated token
# before the first `(`, `[`, or `;` on the SQLITE_API line.
file(STRINGS "${SKEIN_SQLITE_ROOT}/amalgamation/sqlite3.h"
     _skein_api_lines REGEX "^SQLITE_API")
set(_skein_exports "")
foreach(_line IN LISTS _skein_api_lines)
  string(REGEX REPLACE "[ \t]*[([;].*$" "" _stripped "${_line}")
  if(_stripped MATCHES "[* \t]([A-Za-z_][A-Za-z0-9_]*)$")
    list(APPEND _skein_exports "${CMAKE_MATCH_1}")
  endif()
endforeach()

# Non-SQLITE_API C entry points a JNI shim (skein-e2ki) must be able to reach.
# SQLCipher's `sqlite3_key*`/`sqlite3_rekey*` are already `SQLITE_API` in the
# amalgamation header, so they're picked up by the loop above. This tail
# lists only the symbols the header does not declare: SQLCipher's C init
# hook, sqlite-vec's C init entry point, and our own EXTRA_INIT wrapper.
# `vec_version` and `sqlcipher_export` are SQL-registered functions rather
# than C symbols; they are not listed here because a version script can only
# export names present in the objects.
list(APPEND _skein_exports
  sqlcipher_extra_init
  sqlite3_vec_init
  skein_extra_init
)

# Deterministic export set: dedupe + sort. Without this the .so bytes would
# depend on the order awk/grep happens to see lines, breaking
# reproducible-build discipline (see docs/design/AMALGAMATION_POLICY.md).
list(REMOVE_DUPLICATES _skein_exports)
list(SORT _skein_exports)

# Emit the linker version script. `local: *;` explicitly hides every symbol
# not named in `global:` — the actual defense against leaking internals.
set(_skein_exports_body "")
foreach(_sym IN LISTS _skein_exports)
  string(APPEND _skein_exports_body "    ${_sym};\n")
endforeach()
file(WRITE "${_SKEIN_EXPORTS_LD}"
"# GENERATED by native/sqlite/CMakeLists.txt (skein-6rwv). Do not hand-edit.
# Regenerated at every CMake configure from amalgamation/sqlite3.h + tail.
SKEIN_SQLITE_1.0 {
  global:
${_skein_exports_body}  local:
    *;
};
")

list(LENGTH _skein_exports _skein_export_count)
message(STATUS "skein_sqlite: version script exports ${_skein_export_count} symbols -> ${_SKEIN_EXPORTS_LD}")

target_link_options(skein_sqlite PRIVATE
  # Determinism: no build-id note in the .so, no path leakage in DT_RUNPATH.
  "-Wl,--build-id=none"
  "-Wl,--no-rosegment"
  # 16 KiB ELF LOAD-segment alignment (bd skein-gg11.5 / skein-xth5, PP-48,
  # OfflineLLM row OL-46) — see native/llama/CMakeLists.txt §7a for the full
  # rationale for these two explicit flags over
  # `-DANDROID_SUPPORT_FLEXIBLE_PAGE_SIZES=ON`. Applied here too, even though
  # `tools/ci/elf-alignment.sh` only globs for `libskein_sqlite_jni.so`: this
  # library is also `dlopen`'d (as `libskein_sqlite_jni.so`'s `DT_NEEDED`
  # dependency) on the same 16 KiB-page device, so it needs the same
  # alignment to load at all.
  "-Wl,-z,max-page-size=16384"
  "-Wl,-z,common-page-size=16384"
  # Whitelist the exported symbol set; `local: *;` in the script hides
  # everything else. Kept alongside `-fvisibility=hidden` as defense in depth:
  # even a helper accidentally emitted with default visibility stays local.
  "-Wl,--version-script,${_SKEIN_EXPORTS_LD}"
  # sqlite3.h declares APIs (extension loader, preupdate/session hooks,
  # normalized SQL, carray, column origin, mutex_held debug helpers, etc.)
  # whose definitions we deliberately compile out via feature flags
  # (SQLITE_OMIT_LOAD_EXTENSION, no SQLITE_ENABLE_PREUPDATE_HOOK, no
  # SQLITE_DEBUG, etc.). Their names still appear in the header-grep-driven
  # whitelist. `--undefined-version` (LLD's older, permissive default that
  # Android NDK 27+ replaced with --no-undefined-version) tells the linker
  # to skip a whitelist entry whose symbol wasn't actually emitted, instead
  # of erroring out. The exported set is still bounded by "intersection of
  # whitelist and defined symbols", so nothing accidentally leaks.
  "-Wl,--undefined-version"
)

set_target_properties(skein_sqlite PROPERTIES
  C_VISIBILITY_PRESET hidden
  LINK_DEPENDS "${_SKEIN_EXPORTS_LD}"
)

target_link_libraries(skein_sqlite PRIVATE crypto_static log android)

# AGP looks for `.so` files under `<cxxBuildDir>/<abi>/`. That's the default;
# emit `libskein_sqlite.so` there so `packageFossDebug` picks it up
# automatically for the flavor's abiFilters.
set_target_properties(skein_sqlite PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}"
)

# -----------------------------------------------------------------------------
# 6. libskein_sqlite_jni.so — androidx.sqlite-shaped JNI shim (skein-e2ki)
#
# Small (~350 LoC) translation unit at `androidx-jni/skein_jni.c` that maps the
# native methods on `app.skein.core.vault.db.SkeinSQLiteNative` onto the
# corresponding sqlite3_* calls exported by libskein_sqlite.so (skein-6rwv).
# Emitted as a *separate* .so so we don't perturb the reproducibility hash of
# skein_sqlite itself; the .apk carries both.
#
# Reproducibility contract mirrors skein_sqlite: -fno-ident,
# -ffile-prefix-map, -Wl,--build-id=none. No __DATE__/__TIME__ (guarded by
# -Werror=date-time via the amalgamation's rule set; the shim doesn't
# reference either).
# -----------------------------------------------------------------------------
add_library(skein_sqlite_jni SHARED
  "${SKEIN_SQLITE_ROOT}/androidx-jni/skein_jni.c"
)

target_include_directories(skein_sqlite_jni PRIVATE
  "${SKEIN_SQLITE_ROOT}/amalgamation"
)

# Link against the shared skein_sqlite target so the JNI shim resolves
# sqlite3_open / sqlite3_key_v2 / sqlite3_prepare_v2 / etc. dynamically.
# CMake handles both the -lskein_sqlite link line and the DT_NEEDED entry.
target_link_libraries(skein_sqlite_jni PRIVATE skein_sqlite log)

target_compile_options(skein_sqlite_jni PRIVATE
  -O2
  -fvisibility=hidden
  -fno-ident
  "-ffile-prefix-map=${SKEIN_SQLITE_ROOT}=/skein/native/sqlite"
  "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=/skein/build"
  -Wdate-time
  -Werror=date-time
)

target_link_options(skein_sqlite_jni PRIVATE
  "-Wl,--build-id=none"
  "-Wl,--no-rosegment"
  # 16 KiB ELF LOAD-segment alignment (bd skein-gg11.5 / skein-xth5, PP-48,
  # OfflineLLM row OL-46) — see native/llama/CMakeLists.txt §7a for the
  # rationale. This is the library tools/ci/elf-alignment.sh checks by name.
  "-Wl,-z,max-page-size=16384"
  "-Wl,-z,common-page-size=16384"
)

set_target_properties(skein_sqlite_jni PROPERTIES
  C_VISIBILITY_PRESET default
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}"
)
